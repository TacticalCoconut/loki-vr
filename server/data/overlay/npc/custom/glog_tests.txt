// getmapflag of the skill_damage flag: the flag itself, the four rates and the caster of the every-skill adjustment
g_room2,104,106,4	script	GLFlagGet	4_M_01,{
	setarray .@m$[0],"g_room1-1","g_room1-2","g_room2";
	for (.@i = 0; .@i < 3; .@i++)
		dispbottom "sdmg " + .@m$[.@i] + ": " + getmapflag(.@m$[.@i], mf_skill_damage) + " | " + getmapflag(.@m$[.@i], mf_skill_damage, SKILLDMG_PC) + " " + getmapflag(.@m$[.@i], mf_skill_damage, SKILLDMG_MOB) + " " + getmapflag(.@m$[.@i], mf_skill_damage, SKILLDMG_BOSS) + " " + getmapflag(.@m$[.@i], mf_skill_damage, SKILLDMG_OTHER) + " | " + getmapflag(.@m$[.@i], mf_skill_damage, SKILLDMG_CASTER) + " | " + getmapflag(.@m$[.@i], mf_skill_damage, SKILLDMG_MAX);
	end;
}

// setmapflag of the skill_damage flag: rAthena writes args.skill_damage.rate[type] and leaves the caster 0, so
// map_setmapflag_sub refuses ("Skill damage adjustment without casting type") and nothing changes; without both
// values the command fails ("flag data is missing")
g_room2,106,106,4	script	GLFlagSet	4_M_01,{
	setmapflag "g_room2", mf_skill_damage, 50, SKILLDMG_MOB;
	setmapflag "g_room1-1", mf_skill_damage, 77, SKILLDMG_MOB;
	setmapflag "g_room1-2", mf_skill_damage, 88, SKILLDMG_PC;
	setmapflag "g_room2", mf_skill_damage;
	setmapflag "g_room2", mf_skill_damage, 50;
	dispbottom "sdmg set done";
	end;
}

// removemapflag: damage_adjust = {}, skill_damage.clear()
g_room2,108,106,4	script	GLFlagDel	4_M_01,{
	removemapflag "g_room1-2", mf_skill_damage;
	dispbottom "sdmg removed";
	end;
}
